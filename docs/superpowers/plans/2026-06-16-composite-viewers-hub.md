# Composite Viewers Hub Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a standalone, serverless gh-pages hub where a curated set of v2ecoli composites is each viewable through bigraph-loom, bigraph-viz2, and bigraph-viz, all fed from one cached state per composite and regenerated by a single command.

**Architecture:** A data-driven manifest (`docs/viewers/viewers.json`) lists the curated composites. `scripts/regenerate_viewers.py` resolves each composite to a cached state JSON (reusing the dashboard's `_composite_resolve_data`), pre-renders a bigraph-viz SVG and a self-contained bigraph-viz2 HTML, copies the shared bigraph-loom static bundle, and regenerates a static hub `index.html`. `scripts/publish_viewers.sh` pushes `docs/viewers/` to the gh-pages branch. The old `baseline-viewer/` becomes a redirect into the hub.

**Tech Stack:** Python 3.11 (run via `.venv/bin/python`, `PYTHONPATH=.`), pytest, `vivarium_dashboard.server._composite_resolve_data`, `bigraph_viz.plot_bigraph`, `bigraph_viz2.emit_html`, `bigraph_loom` `_dist/` bundle, bash + git worktree for publish.

**Spec:** `docs/superpowers/specs/2026-06-16-composite-viewers-hub-design.md`

---

## Refinement vs spec

The spec proposed a client-side viz2 render shell (`viz2/render.html?state=`). This plan instead **pre-renders** each composite's viz2 view via `bigraph_viz2.emit_html(state)` to `docs/viewers/viz2/<slug>.html`. Rationale: `emit_html` produces a fully self-contained snippet (no runtime fetch, no IIFE-bundle copy to keep in sync), and it is still single-sourced — regen reads the one `data/<slug>.state.json` and emits the HTML. Loom remains runtime-`?stateUrl=`-driven (it has no Python pre-render path).

## File Structure

```
docs/viewers/
  viewers.json                  Create — curated manifest (hand-edited)
  index.html                    Generated by regen — the hub page
  loom/                         Copied by regen — shared bigraph-loom _dist bundle
  data/<slug>.state.json        Written by regen — cached state (feeds loom)
  data/<slug>.view.json         Committed (optional) — saved loom arrangement; regen never overwrites
  viz2/<slug>.html              Written by regen — self-contained bigraph-viz2 snippet
  img/<slug>.svg                Written by regen — bigraph-viz static render
scripts/regenerate_viewers.py   Create — the one regen command (pure functions + main)
scripts/publish_viewers.sh      Create — surgical gh-pages push
tests/test_regenerate_viewers.py Create — unit tests (stubbed resolver/renderers)
docs/baseline-viewer/index.html Modify — replace bundle with a redirect into the hub
docs/baseline-viewer/REGENERATE.md Modify — point at the new regen flow
```

`regenerate_viewers.py` is organized as small pure functions (`load_manifest`,
`hub_html`, `loom_url`, `write_state`, `render_viz_svg`, `render_viz2_html`,
`copy_loom_bundle`) plus a `main()` that wires them. The pure functions are
unit-tested; composite resolution and SVG/HTML rendering take **injectable
callables** so tests run without heavy deps.

```python
#!/usr/bin/env python
"""Regenerate the composite viewers hub (docs/viewers/).

For each composite in docs/viewers/viewers.json this resolves the composite to a
cached state JSON, pre-renders a bigraph-viz SVG and a self-contained
bigraph-viz2 HTML snippet, copies the shared bigraph-loom static bundle, and
regenerates the hub index.html. Unresolvable composites are skipped with a
warning (never fatal).

Run locally (the ParCa cache must be on disk so heavy composites resolve):
    PYTHONPATH=. .venv/bin/python scripts/regenerate_viewers.py
"""
from __future__ import annotations

import json
import shutil
import sys
from dataclasses import dataclass
from pathlib import Path

REPO_ROOT = Path(__file__).resolve().parent.parent
VIEWERS_DIR = REPO_ROOT / "docs" / "viewers"


@dataclass
class Entry:
    slug: str
    id: str
    title: str
    blurb: str = ""


def load_manifest(path: Path) -> list[Entry]:
    raw = json.loads(Path(path).read_text(encoding="utf-8"))
    return [Entry(slug=e["slug"], id=e["id"], title=e["title"],
                  blurb=e.get("blurb", "")) for e in raw]


def write_state(state: dict, slug: str, data_dir: Path) -> Path:
    data_dir.mkdir(parents=True, exist_ok=True)
    out = data_dir / f"{slug}.state.json"
    # Wrap as {state: ...} — the shape bigraph-loom's ?stateUrl= reader unwraps
    # (matches the existing docs/baseline-viewer/data/baseline.state.json).
    out.write_text(json.dumps({"state": state}, indent=2), encoding="utf-8")
    return out
```

```python
def loom_url(entry: Entry, *, has_view: bool) -> str:
    url = (f"loom/index.html?static=1&id={entry.id}"
           f"&stateUrl=../data/{entry.slug}.state.json")
    if has_view:
        url += f"&viewUrl=../data/{entry.slug}.view.json"
    return url


def _esc(s: str) -> str:
    return (s.replace("&", "&amp;").replace("<", "&lt;")
             .replace(">", "&gt;").replace('"', "&quot;"))


def hub_html(rows: list[dict]) -> str:
    """rows: [{entry: Entry, has_view: bool, has_viz2: bool, has_svg: bool}]."""
    cards = []
    for r in rows:
        e = r["entry"]
        buttons = [f'<a class="btn" href="{_esc(loom_url(e, has_view=r["has_view"]))}">Loom</a>']
        if r["has_viz2"]:
            buttons.append(f'<a class="btn" href="viz2/{e.slug}.html">Viz2</a>')
        if r["has_svg"]:
            buttons.append(f'<a class="btn" href="img/{e.slug}.svg">Viz</a>')
        cards.append(
            '<div class="card">'
            f'<h2>{_esc(e.title)}</h2>'
            f'<p class="blurb">{_esc(e.blurb)}</p>'
            f'<p class="id"><code>{_esc(e.id)}</code></p>'
            f'<div class="btns">{"".join(buttons)}</div>'
            '</div>')
    return _PAGE_TEMPLATE.replace("{{CARDS}}", "\n".join(cards))
```

```python
def resolve_state_via_dashboard(spec_id: str) -> dict | None:
    """Resolve a composite to its loom state dict by reusing the dashboard's
    pure resolver. Returns None on any failure (logged by caller)."""
    import vivarium_dashboard.server as srv
    srv.WORKSPACE = REPO_ROOT
    data = srv._composite_resolve_data(spec_id)
    if not data or not isinstance(data.get("state"), dict):
        return None
    return data["state"]


def render_viz_svg(state: dict, slug: str, img_dir: Path) -> Path | None:
    """Best-effort bigraph-viz static SVG. Returns the path or None on failure."""
    try:
        from bigraph_viz import plot_bigraph
        img_dir.mkdir(parents=True, exist_ok=True)
        # plot_bigraph writes <out_dir>/<filename>.<file_format>; returns a graph obj.
        plot_bigraph(state, out_dir=str(img_dir), filename=slug,
                     file_format="svg", show_compiled_state=False)
        out = img_dir / f"{slug}.svg"
        return out if out.is_file() else None
    except Exception as exc:  # noqa: BLE001
        print(f"  ! viz SVG skipped for {slug}: {exc}", file=sys.stderr)
        return None


def render_viz2_html(state: dict, slug: str, viz2_dir: Path) -> Path | None:
    """Best-effort self-contained bigraph-viz2 snippet. Returns path or None."""
    try:
        from bigraph_viz2 import emit_html
        viz2_dir.mkdir(parents=True, exist_ok=True)
        html = emit_html(state, height="100vh")
        out = viz2_dir / f"{slug}.html"
        out.write_text(html, encoding="utf-8")
        return out
    except Exception as exc:  # noqa: BLE001
        print(f"  ! viz2 skipped for {slug}: {exc}", file=sys.stderr)
        return None


def build_rows(entries, *, VIEWERS_DIR, resolve, render_svg, render_viz2):
    data_dir = VIEWERS_DIR / "data"
    img_dir = VIEWERS_DIR / "img"
    viz2_dir = VIEWERS_DIR / "viz2"
    rows = []
    for e in entries:
        print(f"- {e.slug} ({e.id})")
        state = resolve(e.id)
        if state is None:
            print(f"  ! skipped: {e.id} did not resolve", file=sys.stderr)
            continue
        write_state(state, e.slug, data_dir)
        svg = render_svg(state, e.slug, img_dir)
        viz2 = render_viz2(state, e.slug, viz2_dir)
        rows.append({
            "entry": e,
            "has_view": (data_dir / f"{e.slug}.view.json").is_file(),
            "has_svg": svg is not None,
            "has_viz2": viz2 is not None,
        })
    return rows
```

```python
def loom_dist_dir() -> Path:
    """Locate the installed bigraph-loom static build (_dist/)."""
    import bigraph_loom
    return Path(bigraph_loom.__file__).resolve().parent / "_dist"


def copy_loom_bundle(dest: Path, *, src: Path | None = None) -> None:
    src = src or loom_dist_dir()
    if dest.exists():
        shutil.rmtree(dest)
    shutil.copytree(src, dest)


def main() -> int:
    entries = load_manifest(VIEWERS_DIR / "viewers.json")
    print(f"Regenerating {len(entries)} composite viewers into {VIEWERS_DIR}")
    rows = build_rows(entries, VIEWERS_DIR=VIEWERS_DIR,
                      resolve=resolve_state_via_dashboard,
                      render_svg=render_viz_svg, render_viz2=render_viz2_html)
    copy_loom_bundle(VIEWERS_DIR / "loom")
    (VIEWERS_DIR / "index.html").write_text(hub_html(rows), encoding="utf-8")
    print(f"Done: {len(rows)}/{len(entries)} composites in the hub "
          f"({VIEWERS_DIR / 'index.html'})")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```
